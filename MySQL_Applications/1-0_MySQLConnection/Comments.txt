How to connect to the MySQL Database using Lazarus and Zeos Componnent

ON MYSQL
Accesses MySQL client
#sudo mysql

Create your database
mysql> CREATE DATABASE APPLICATION_DB;
mysql> USE APPLICATION_DB;

Create user
CREATE USER 'admin'@'localhost' IDENTIFIED BY '1234';
GRANT ALL PRIVILEGES ON APPLICATION_DB.* TO 'admin'@'localhost';
FLUSH PRIVILEGES;


Create your table
CREATE TABLE SENSORS(
ID INT PRIMARY KEY AUTO_INCREMENT,
DESCRIPTION VARCHAR(30),
VALUE VARCHAR(10)
);

Insert values on database table
INSERT INTO SENSORS VALUES(NULL, 'MOTOR_SPEED', '253');
INSERT INTO SENSORS VALUES(NULL, 'INTERNAL_PRESSURE', '157.8');
INSERT INTO SENSORS VALUES(NULL, 'KEY_START', '1');

ON LAZARUS
ZConnection componnent proprieties
Protocol: mysql
HostBame: localhost
Port: 3306
User: admin
Password: 1234
Database: APPLICATION_DB
LibraryLocation*: /usr/lib/x86_64-linux-gnu/libmysqlclient.so
